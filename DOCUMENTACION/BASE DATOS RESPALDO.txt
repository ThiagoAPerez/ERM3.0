📘 OpenAPI 3.0 — EL RAPIDÍN (Backend v1)
openapi: 3.0.3
info:
  title: EL RAPIDÍN API
  version: "1.0.0"
  description: Backend v1 - Contrato oficial

servers:
  - url: https://api.elrapidin.com/v1

tags:
  - name: Auth
  - name: Users
  - name: Businesses
  - name: Products
  - name: Orders
  - name: Delivery
  - name: Ratings
  - name: Admin

components:
  securitySchemes:
    BearerAuth:
      type: http
      scheme: bearer
      bearerFormat: JWT

  schemas:
    User:
      type: object
      properties:
        id: { type: integer }
        name: { type: string }
        phone: { type: string }
        email: { type: string, nullable: true }
        role:
          type: string
          enum: [ADMIN, CLIENT, BUSINESS, DELIVERY]
        status:
          type: string
          enum: [ACTIVE, INACTIVE, SUSPENDED]

    LoginRequest:
      type: object
      required: [phone, password]
      properties:
        phone: { type: string }
        password: { type: string }

    LoginResponse:
      type: object
      properties:
        accessToken: { type: string }
        user: { $ref: "#/components/schemas/User" }

    ProductItem:
      type: object
      properties:
        productId: { type: integer }
        quantity: { type: integer }

    OrderCreateRequest:
      type: object
      required: [providerType, providerId, items, delivery, paymentMethod]
      properties:
        providerType:
          type: string
          enum: [BUSINESS, STORE]
        providerId: { type: integer }
        items:
          type: array
          items: { $ref: "#/components/schemas/ProductItem" }
        delivery:
          type: object
          required: [address, municipality, contactName, contactPhone]
          properties:
            address: { type: string }
            municipality: { type: string }
            contactName: { type: string }
            contactPhone: { type: string }
        paymentMethod:
          type: string
          enum: [CASH, PSE]

    OrderResponse:
      type: object
      properties:
        orderId: { type: integer }
        orderCode: { type: string }
        status: { type: string }
        estimatedPickupAt: { type: string, format: date-time }
        estimatedDeliveryAt: { type: string, format: date-time }
        totalPrice: { type: number }

security:
  - BearerAuth: []
